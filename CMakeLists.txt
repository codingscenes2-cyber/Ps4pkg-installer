cmake_minimum_required(VERSION 3.20)
project(PKGMasterCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CURL QUIET)
find_package(Threads REQUIRED)
add_library(pkgmaster_core
 src/core/job.cpp src/core/queue.cpp src/core/queue_store.cpp src/core/url_policy.cpp src/core/state_transition.cpp
 src/downloader/range_plan.cpp src/downloader/range_file_assembler.cpp src/downloader/resume_manifest.cpp src/downloader/http_downloader.cpp src/downloader/http_probe.cpp src/downloader/parallel_range_downloader.cpp src/downloader/download_coordinator.cpp src/platform/installer_bridge.cpp src/platform/capabilities.cpp)
target_include_directories(pkgmaster_core PUBLIC include)
target_link_libraries(pkgmaster_core PUBLIC Threads::Threads)
if(CURL_FOUND)
 target_compile_definitions(pkgmaster_core PRIVATE PKGMASTER_HAS_CURL=1)
 target_link_libraries(pkgmaster_core PUBLIC CURL::libcurl)
 message(STATUS "PKGMaster HTTP transport enabled with libcurl")
else()
 message(WARNING "libcurl not found: HTTP downloader will report Unsupported")
endif()
add_executable(pkgmaster_core_demo src/main.cpp)
target_link_libraries(pkgmaster_core_demo PRIVATE pkgmaster_core)
if(MSVC)
 target_compile_options(pkgmaster_core PRIVATE /W4)
else()
 target_compile_options(pkgmaster_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(PKGMASTER_BUILD_TESTS "Build host-side tests" ON)
if(PKGMASTER_BUILD_TESTS)
 enable_testing()
 add_executable(pkgmaster_tests tests/core_tests.cpp)
 target_link_libraries(pkgmaster_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_core_tests COMMAND pkgmaster_tests)
 add_executable(pkgmaster_downloader_tests tests/downloader_tests.cpp)
 target_link_libraries(pkgmaster_downloader_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_downloader_tests COMMAND pkgmaster_downloader_tests)
 add_executable(pkgmaster_http_probe_tests tests/http_probe_tests.cpp)
 target_link_libraries(pkgmaster_http_probe_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_http_probe_tests COMMAND pkgmaster_http_probe_tests)
 add_executable(pkgmaster_parallel_range_tests tests/parallel_range_tests.cpp)
 target_link_libraries(pkgmaster_parallel_range_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_parallel_range_tests COMMAND pkgmaster_parallel_range_tests)
 add_executable(pkgmaster_download_metrics_tests tests/download_metrics_tests.cpp)
 target_link_libraries(pkgmaster_download_metrics_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_download_metrics_tests COMMAND pkgmaster_download_metrics_tests)
 add_executable(pkgmaster_download_coordinator_tests tests/download_coordinator_tests.cpp)
 target_link_libraries(pkgmaster_download_coordinator_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_download_coordinator_tests COMMAND pkgmaster_download_coordinator_tests)
 add_executable(pkgmaster_resume_tests tests/resume_manifest_tests.cpp)
 target_link_libraries(pkgmaster_resume_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_resume_tests COMMAND pkgmaster_resume_tests)
 add_executable(pkgmaster_range_file_tests tests/range_file_assembler_tests.cpp)
 target_link_libraries(pkgmaster_range_file_tests PRIVATE pkgmaster_core)
 add_test(NAME pkgmaster_range_file_tests COMMAND pkgmaster_range_file_tests)
endif()
